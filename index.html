// script.js // Instructions: // - Add new game HTML files to the games/ folder. // - Copy games/game-template.html to create a new file (e.g., games/mygame.html). // - Then add a new object to the `games` array below with { id, title, file }. // - id must be unique // - file should be a relative path like "games/mygame.html" // - The hub will load the game HTML inside an iframe when you click Play. // - The UI does NOT display the file path or URL. const games = [ { id: "ttt", title: "Tic Tac Toe (example)", desc: "One-file Tic Tac Toe. Copy its format to create other games.", file: "games/tictactoe.html" }, { id: "template", title: "Game Template (copy this)", desc: "A commented template you can copy for new games.", file: "games/game-template.html" } ]; // --- UI rendering --- const grid = document.getElementById('grid'); const modal = document.getElementById('modal'); const player = document.getElementById('player'); const closeBtn = document.getElementById('closeBtn'); const search = document.getElementById('search'); function renderGames(list){ grid.innerHTML = ''; list.forEach(g=>{ const card = document.createElement('article'); card.className = 'card'; card.innerHTML = `

${escapeHtml(g.title)}

${escapeHtml(g.desc || '')}

`; grid.appendChild(card); }); // attach listeners document.querySelectorAll('.btn').forEach(btn=>{ btn.addEventListener('click', e=>{ const file = decodeURIComponent(btn.getAttribute('data-file') || ''); if(!file) return; openModalWith(file); }); }); } function openFolder(fileEnc){ const file = decodeURIComponent(fileEnc); // "Info" opens the template in the modal too; we purposely do not display the URL anywhere. openModalWith(file); } function openModalWith(file){ player.src = file; modal.setAttribute('aria-hidden','false'); // focus inside iframe when loaded } closeBtn.addEventListener('click', ()=>{ modal.setAttribute('aria-hidden','true'); player.src = 'about:blank'; }); modal.addEventListener('click', (e)=>{ if(e.target === modal) { closeBtn.click(); } }); search.addEventListener('input', ()=>{ const q = search.value.trim().toLowerCase(); const filtered = games.filter(g => (g.title + ' ' + (g.desc||'')).toLowerCase().includes(q)); renderGames(filtered); }); // initial render renderGames(games); // tiny helper to avoid injection in titles/descriptions function escapeHtml(s){ return String(s || '').replace(/[&<>"']/g, (c)=> ({ '&':'&','<':'<','>':'>','"':'"',"'":''' })[c]); }
